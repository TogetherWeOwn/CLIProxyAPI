"""Retained read-only CLIProxy probes (no secrets printed)."""
import json
import sys
import urllib.error
import urllib.request

import yaml

import os
cfg = yaml.safe_load(open(os.environ.get('CLIPROXY_CFG', '/path/to/stack/config.yaml')))
KEY = cfg['api-keys'][0]
BASE = os.environ.get('CLIPROXY_BASE', 'http://example.invalid')


def req(path, body=None, timeout=120):
    hdrs = {'Authorization': 'Bearer ' + KEY} if path == '/v1/models' else {
        'Authorization': 'Bearer ' + KEY, 'x-api-key': KEY, 'Content-Type': 'application/json', 'anthropic-version': '2023-06-01'}
    r = urllib.request.Request(BASE + path, data=json.dumps(body).encode() if body is not None else None, headers=hdrs)
    try:
        with urllib.request.urlopen(r, timeout=timeout) as x:
            return x.status, json.loads(x.read() or b'null')
    except urllib.error.HTTPError as e:
        return e.code, e.read()[:300].decode('utf-8', 'replace')
    except Exception as e:
        return 0, type(e).__name__


what = sys.argv[1] if len(sys.argv) > 1 else 'models'
if what == 'models':
    s, d = req('/v1/models')
    ids = [m['id'] for m in (d or {}).get('data', [])] if s == 200 else []
    muse = sorted(i for i in ids if 'muse' in i)
    print(json.dumps({'status': s, 'total': len(ids), 'muse_bare': [i for i in muse if '/' not in i],
                      'muse_prefixed': [i for i in muse if i.startswith('meta-muse/')]}))
elif what == 'models-all':
    s, d = req('/v1/models')
    ids = [m['id'] for m in (d or {}).get('data', [])] if s == 200 else []
    print(json.dumps({'status': s, 'ids': ids}))
elif what == 'stream':
    model = sys.argv[2]
    body = {'model': model, 'max_tokens': 64, 'stream': True, 'messages': [{'role': 'user', 'content': 'Say OK.'}]}
    r = urllib.request.Request(BASE + '/v1/messages', data=json.dumps(body).encode(), headers={'Authorization': 'Bearer ' + KEY, 'x-api-key': KEY, 'Content-Type': 'application/json', 'anthropic-version': '2023-06-01'})
    try:
        with urllib.request.urlopen(r, timeout=120) as x:
            data = x.read().decode('utf-8', 'replace')
            print(json.dumps({'status': x.status, 'events': data.count('event:'), 'stop': 'message_stop' in data}))
    except urllib.error.HTTPError as e:
        print(json.dumps({'status': e.code, 'error': e.read()[:200].decode('utf-8', 'replace')}))
    except Exception as e:
        print(json.dumps({'status': 0, 'error': type(e).__name__}))
elif what == 'tool':
    model = sys.argv[2]
    tools = [{'name': 'get_weather', 'description': 'Get weather for a city',
              'input_schema': {'type': 'object', 'properties': {'city': {'type': 'string'}}, 'required': ['city']}}]
    msgs = [{'role': 'user', 'content': 'What is the weather in Paris? Use the tool.'}]
    s, d = req('/v1/messages', {'model': model, 'max_tokens': 256, 'tools': tools, 'messages': msgs})
    out = {'status': s}
    if s == 200:
        out['stop_reason'] = d.get('stop_reason')
        tu = [b for b in d.get('content', []) if b.get('type') == 'tool_use']
        out['tool_use'] = [b['name'] for b in tu]
        if len(sys.argv) > 3 and sys.argv[3] == 'multiturn' and tu:
            msgs2 = msgs + [{'role': 'assistant', 'content': d['content']},
                            {'role': 'user', 'content': [{'type': 'tool_result', 'tool_use_id': tu[0]['id'], 'content': 'Sunny, 21C'}]}]
            s2, d2 = req('/v1/messages', {'model': model, 'max_tokens': 256, 'tools': tools, 'messages': msgs2})
            out['turn2_status'] = s2
            if s2 == 200:
                out['turn2_stop'] = d2.get('stop_reason')
                tu2 = [b for b in d2.get('content', []) if b.get('type') == 'tool_use']
                if tu2:
                    msgs3 = msgs2 + [{'role': 'assistant', 'content': d2['content']},
                                     {'role': 'user', 'content': [{'type': 'tool_result', 'tool_use_id': tu2[0]['id'], 'content': 'ok'}]}]
                    s3, _ = req('/v1/messages', {'model': model, 'max_tokens': 128, 'tools': tools, 'messages': msgs3})
                    out['turn3_status'] = s3
    else:
        out['error'] = str(d)[:200]
    print(json.dumps(out))
