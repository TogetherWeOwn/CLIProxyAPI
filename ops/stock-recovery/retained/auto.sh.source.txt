#!/usr/bin/env bash
# CLIProxyAPI v8.0.58-stock -> official v8.0.60 image swap (owner 2026-11-07 "also new cliproxy update"). Detached, self-testing, auto-rollback
# (owner rule 2026-11-04: anything that can cut the operator's model route runs as ONE detached script that
# tests itself and rolls back on its own; owner 2026-11-07: "make sure ... you have a way back in").
# Forward path changes ONLY the compose image line. Rollback restores today's exact compose/config and starts the old
# image. kv.db is NOT auto-restored: v8.0.60 may rotate OAuth refresh tokens during the test window, and restoring an
# older copy would replay a used refresh token (revocation risk). A cold copy taken while stopped is kept in $SD for manual use.
#
# Launch:  setsid nohup <path>/auto.sh run </dev/null >><state>/auto.log 2>&1 &
# Manual way back in (no model, no network needed):  STATE_DIR=<state dir> <path>/auto.sh rollback
# Env: S C SVC PORT STATE_DIR IMG_NEW IMG_OLD_ID PLUGINS REQUIRED_MODELS PROBE_MODELS REPLAY_MODELS
#      LIVE_PROBES=1|0 FAIL_AT=poststart|hang WD_SECS (scratch/staging drills only change these)
set -uo pipefail
W=$(dirname "$(readlink -f "$0")")
S=${S:-/path/to/stack}
C=${C:-cliproxy}
SVC=${SVC:-cli-proxy-api}
PORT=${PORT:-8317}
SD=${STATE_DIR:-$W/state-prod}
IMG_NEW=${IMG_NEW:-cliproxy-local:v8.0.60-stock}
IMG_NEW_ID=${IMG_NEW_ID:-sha256:873ec3df3b673a82f7d956d4636c7736b0c961dc6ee28f9fb403238b2898dc0d}
PLUGINS=${PLUGINS:-"cloudflare-workers-ai hopper meta nvidia-nim openrouter synthetic"}
REQUIRED_MODELS=${REQUIRED_MODELS-"gpt-6.1-sol muse-spark-1.3-contributor muse-wonder-1.0 muse-wonder-full-1.0-contributor"}
PROBE_MODELS=${PROBE_MODELS-"claude-fable-5-dd-5-5-supo-edualc gpt-6.1-sol"}
REPLAY_MODELS=${REPLAY_MODELS-"muse-spark-1.3-contributor muse-wonder-1.0 muse-spark-1.3-contributor"}
LIVE_PROBES=${LIVE_PROBES:-1}
FAIL_AT=${FAIL_AT:-}
WD_SECS=${WD_SECS:-900}
export CLIPROXY_CFG=$S/config.yaml CLIPROXY_BASE=http://example.invalid
mkdir -p "$SD"; chmod 700 "$SD"
log() { echo "$(date -u +%H:%M:%SZ) [$$] $*"; }
state() { echo "$1" > "$SD/state"; log "STATE $1"; }
dc() { (cd "$S" && timeout 150 docker compose "$@"); }
probe() { timeout 200 python3 "$W/probe.py" "$@"; }

models_ok() {  # required model ids present in /v1/models
  probe models-all > "$SD/models.json" 2>/dev/null || return 1
  python3 - "$SD/models.json" "$REQUIRED_MODELS" <<'EOF'
import json, sys
d = json.load(open(sys.argv[1])); want = sys.argv[2].split()
miss = [m for m in want if m not in d['ids']]
print('models', d['status'], len(d['ids']), 'missing', miss)
sys.exit(0 if d['status'] == 200 and not miss else 1)
EOF
}

live_ok() {  # real tool-call round trip (streaming + non-streaming) on our own routes
  [ "$LIVE_PROBES" = 1 ] || return 0
  local m
  for m in $PROBE_MODELS; do
    probe tool "$m" multiturn | tee -a "$SD/probes.log" | grep -c '"turn2_status": 200' >/dev/null || { log "probe red: $m"; return 1; }
    probe stream "$m" | tee -a "$SD/probes.log" | grep -c '"status": 200' >/dev/null || { log "stream probe red: $m"; return 1; }
  done
}

disabled_count() { sudo -n python3 -I "$W/authcount.py" "$S/auths"; }

rollback() {
  exec 9>"$SD/rollback.lock"; flock -n 9 || { log "rollback already running elsewhere"; return 0; }
  grep -qx rolledback "$SD/state" 2>/dev/null && return 0
  state rolling-back
  local TS; TS=$(cat "$SD/ts")
  dc stop "$SVC" || docker stop "$C" || true
  cp "$SD/compose.yaml.orig" "$S/compose.yaml"
  cp "$SD/config.yaml.orig" "$S/config.yaml"; chmod 600 "$S/config.yaml"
  for attempt in 1 2 3; do
    dc up -d --no-deps "$SVC" || docker start "$C" || true
    for _ in $(seq 1 30); do sleep 3; models_ok >/dev/null 2>&1 && break; done
    if [ "$(docker inspect -f '{{.Image}}' "$C" 2>/dev/null)" = "$(cat "$SD/img_old_id")" ] && models_ok && live_ok; then
      state rolledback; log "rollback verified (attempt $attempt): $(docker inspect -f '{{.Config.Image}} running={{.State.Running}}' "$C")"; return 0
    fi
    log "rollback attempt $attempt not healthy; retrying"; dc stop "$SVC" || true
  done
  state rollback-FAILED; dc up -d --no-deps "$SVC" || docker start "$C" || true; return 1
}

since_logs() {  # host logger -> logs/main*.log; plugin loggers -> container stdout. Search both.
  docker logs --since "$START" "$C" 2>&1
  sudo -n awk -v s="$(date -u -d @"$START" '+%Y-%m-%d %H:%M:%S')" 'substr($0,2,19) >= s' "$S/logs/main.log" "$S"/logs/main-*.log 2>/dev/null
  return 0  # a missing rotated log must not fail the pipeline under pipefail
}

fail() { trap - ERR; log "RED: $*"; [ -n "${START:-}" ] && (umask 077; docker logs --since "$START" "$C" > "$SD/new-container.log" 2>&1); rollback; exit 1; }

if [ "${1:-}" = watchdog ]; then
  sleep "$WD_SECS"
  case "$(cat "$SD/state" 2>/dev/null)" in verified|rolledback|aborted-unchanged|rollback-FAILED) exit 0;; esac
  log "WATCHDOG: state=$(cat "$SD/state") after ${WD_SECS}s; killing main process group and rolling back"
  kill -- -"$(cat "$SD/main.pgid")" 2>/dev/null; sleep 2; rollback; exit 0
fi
if [ "${1:-}" = rollback ]; then rollback; exit $?; fi
[ "${1:-}" = run ] || { echo "usage: auto.sh run|rollback"; exit 2; }
echo $$ > "$SD/main.pid"; ps -o pgid= $$ | tr -d ' ' > "$SD/main.pgid"; rm -f "$SD/state" "$SD/probes.log"
state preflight

# ---- preflight: nothing changed yet; any failure = clean abort ----
abort() { log "ABORT before any change: $*"; state aborted-unchanged; exit 3; }
sudo -n true || abort "sudo -n unavailable"
[ "$(docker inspect -f '{{.State.Running}}' "$C" 2>/dev/null)" = true ] || abort "container not running"
docker inspect -f '{{.Image}}' "$C" > "$SD/img_old_id"
docker inspect -f '{{.Config.Image}}' "$C" > "$SD/img_old_tag"
[ "$(docker image inspect -f '{{.Id}}' "$IMG_NEW" 2>/dev/null)" = "$IMG_NEW_ID" ] || abort "new image missing or id mismatch"
docker image inspect "$(cat "$SD/img_old_id")" >/dev/null 2>&1 || abort "rollback image not local"
grep -q "image: $(cat "$SD/img_old_tag")" "$S/compose.yaml" || abort "compose image line does not match running image"
(cd "$S" && timeout 60 docker compose config -q) || abort "compose config invalid"
sudo -n test -e "$S/auths/kv.db" || abort "kv.db missing (unexpected on v8.0.58)"
models_ok || abort "current prod not healthy"
live_ok || abort "live probe red on current prod"
disabled_count > "$SD/disabled.before" || abort "auth count failed"
log "baseline disabled=$(cat "$SD/disabled.before")"

# ---- backup ----
TS=$(date -u +%Y%m%dT%H%M%SZ); echo "$TS" > "$SD/ts"
cp -p "$S/compose.yaml" "$SD/compose.yaml.orig" && cp -p "$S/config.yaml" "$SD/config.yaml.orig" && chmod 600 "$SD/config.yaml.orig" || abort "orig copies failed"
cp -p "$S/compose.yaml" "$S/compose.yaml.bak-$TS-pre-v8060" || abort "bak copy failed"
sudo -n tar -czf "$S/auths-config-plugins-bak-$TS-pre-v8060.tgz" --exclude=auths/kv.db -C "$S" auths config.yaml compose.yaml plugins \
  && sudo -n chown "$(id -u):$(id -g)" "$S/auths-config-plugins-bak-$TS-pre-v8060.tgz" && chmod 600 "$S/auths-config-plugins-bak-$TS-pre-v8060.tgz" || abort "backup failed"
log "backup $TS ok"

# ---- from here every failure rolls back; watchdog (separate session) covers hangs/kills ----
setsid nohup env S="$S" C="$C" SVC="$SVC" PORT="$PORT" STATE_DIR="$SD" LIVE_PROBES="$LIVE_PROBES" WD_SECS="$WD_SECS" \
  REQUIRED_MODELS="$REQUIRED_MODELS" PROBE_MODELS="$PROBE_MODELS" "$W/auto.sh" watchdog </dev/null >>"$SD/watchdog.log" 2>&1 &
trap 'fail "ERR at line $LINENO"' ERR
state stopping
dc stop "$SVC"
[ "$(docker inspect -f '{{.State.Running}}' "$C")" = false ] || fail "container still running"
sudo -n cp -p "$S/auths/kv.db" "$SD/kv.db.cold-stopped" || fail "kv.db cold copy failed"
sed -i "s#image: $(cat "$SD/img_old_tag")\( .*\)\?\$#image: $IMG_NEW \# v8.0.60 official ($TS). Rollback: STATE_DIR=$SD $W/auto.sh rollback#" "$S/compose.yaml"
grep -q "image: $IMG_NEW " "$S/compose.yaml" || fail "image line not updated"
state starting
START=$(date -u +%s)
dc up -d --no-deps "$SVC"
[ "$FAIL_AT" = hang ] && { log "TEST: hanging"; sleep 100000; }

# ---- post-checks ----
READY='full client load complete|server clients and configuration updated'  # v8.0.51 | v8.0.58 wording
for _ in $(seq 1 50); do sleep 2; since_logs | grep -Ec "$READY" >/dev/null && break; done
since_logs | grep -Ec "$READY" >/dev/null || fail "no full client load within 100s"
[ "$(docker inspect -f '{{.Image}} {{.State.Running}}' "$C")" = "$IMG_NEW_ID true" ] || fail "wrong image or not running"
for _ in $(seq 1 30); do sleep 3; models_ok >/dev/null 2>&1 && break; done
models_ok || fail "required models missing"
for _ in $(seq 1 40); do sleep 3; since_logs | grep -c "plugin registered plugin_id=hopper" >/dev/null && break; done
L=$(since_logs)
for p in $PLUGINS; do
  grep -q "plugin loaded plugin_id=$p " <<<"$L" || fail "plugin $p not loaded"
  grep -q "plugin registered plugin_id=$p" <<<"$L" || fail "plugin $p not registered"
done
grep -Eiq 'panic|probe failed' <<<"$L" && fail "panic or plugin probe failure in logs"
log "plugins loaded+registered: $PLUGINS"
[ "$FAIL_AT" = poststart ] && fail "TEST: injected post-start failure"
live_ok || fail "live probe (operator model route / sol)"
if [ "$LIVE_PROBES" = 1 ] && [ -n "$REPLAY_MODELS" ]; then
  replay=no
  for m in $REPLAY_MODELS; do
    probe tool "$m" multiturn | tee -a "$SD/probes.log" | grep -c '"turn2_status": 200' >/dev/null || continue
    sleep 3
    if since_logs | grep -E 'reasoning replay' | grep -E 'kept=[1-9]' | grep -c 'dropped_foreign=0' >/dev/null; then replay=yes; break; fi
  done
  [ "$replay" = yes ] || fail "no meta replay line kept>=1 dropped_foreign=0"
fi
disabled_count > "$SD/disabled.after"; cmp -s "$SD/disabled.before" "$SD/disabled.after" || fail "disabled credential count changed"
trap - ERR
state verified
log "UPGRADE VERIFIED: $(docker inspect -f '{{.Config.Image}} running={{.State.Running}}' "$C") kv.db=$(sudo -n test -e "$S/auths/kv.db" && echo present || echo absent)"
